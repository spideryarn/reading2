type Page = Awaited<ReturnType<typeof open>>["page"];

async function ready(page: Page, path: string, wait?: string) {
  await page.goto(`${BASE}${path}`, { waitUntil: "domcontentloaded" });
  if (wait) await page.waitForSelector(wait, { timeout: 20_000 }).catch(() => {});
  await page.waitForTimeout(1200);
  await page.addScriptTag({ content: IN_PAGE + MORE });
}

/** Land on the nth match with a real Tab: script focus goes to it only to find
 *  its place in the order, then a real Shift+Tab leaves and a real Tab returns. */
async function tabTo(page: Page, selector: string, nth = 0): Promise<unknown> {
  const exists = await page.evaluate(
    ([sel, n]) => {
      const el = document.querySelectorAll(sel as string)[n as number] as HTMLElement | undefined;
      if (!el) return false;
      el.scrollIntoView({ block: "center" });
      el.focus({ preventScroll: true });
      (window as any).k1target = el;
      return document.activeElement === el;
    },
    [selector, nth] as const,
  );
  if (!exists) return { missingOrUnfocusable: selector };
  await page.keyboard.press("Shift+Tab");
  const left = await page.evaluate(() => document.activeElement !== (window as any).k1target);
  await page.keyboard.press("Tab");
  const hit = await page.evaluate(() => document.activeElement === (window as any).k1target);
  if (!hit) return { notReached: selector, left, at: await page.evaluate(() => document.activeElement?.tagName + "." + String(document.activeElement?.className).slice(0, 50)) };
  await page.waitForTimeout(300);
  const info = await page.evaluate(() => (window as any).k1.focusInfo());
  return { leftThenTabbedBack: left, ...(info as object) };
}
