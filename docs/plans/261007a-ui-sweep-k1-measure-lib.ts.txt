/* K1 browser helpers. Run with `npx tsx` from the worktree root. */
import { chromium } from "/var/tmp/spideryarn-worktrees/agent-a9363ef371ceb477c/node_modules/playwright-core/index.mjs";
import { signIn } from "/var/tmp/spideryarn-worktrees/agent-a9363ef371ceb477c/scripts/browser-sign-in.ts";

export const BASE = process.env.SPIDERYARN_BASE_URL ?? "http://localhost:5273";

export async function open(opts: { width: number; height?: number; touch?: boolean; theme: "dark" | "light" }) {
  const browser = await chromium.launch({
    headless: true,
    executablePath: "/usr/bin/google-chrome-stable",
    args: ["--no-sandbox"],
  });
  const context = await browser.newContext({
    viewport: { width: opts.width, height: opts.height ?? 900 },
    hasTouch: !!opts.touch,
  });
  await context.addInitScript({ content: "globalThis.__name = globalThis.__name || ((f) => f);" });
  await context.addInitScript({
    content: `try { localStorage.setItem("spya.appearance", ${JSON.stringify(opts.theme)}); } catch {}`,
  });
  const page = await context.newPage();
  await signIn(page, BASE);
  return { browser, context, page };
}

  rgb(color) {
    const c = document.createElement("canvas"); c.width = c.height = 1;
    const x = c.getContext("2d", { willReadFrequently: true });
    x.clearRect(0,0,1,1); x.fillStyle = "#000"; x.fillStyle = color; x.fillRect(0,0,1,1);

  lum(c) {
    const f = (v) => { v /= 255; return v <= 0.04045 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4); };
