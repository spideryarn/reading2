/* Blind judging pool for the queries the fallback fires on. For each: every
   block that scored >= 0.45 on any of the three runs, plus as many decoys
   (blocks that never scored above 0.3, picked by a seeded shuffle), in article
   order, with no score and nothing saying which is which. Prints article text:
   scratch only. Writes pool-abstract.md (for the judge) and pool-abstract-key.json. */
import { readFileSync, writeFileSync } from "node:fs";
import { DIR, quickBlocks } from "./lib.ts";
import { QUERIES, load } from "./abstract-run.ts";

const raw = JSON.parse(readFileSync(`${DIR}/abstract-raw.json`, "utf8"));
let seed = 261003;
const rnd = () => ((seed = (seed * 1103515245 + 12345) & 0x7fffffff) / 0x7fffffff);

let md = "";
const key: Record<string, { candidates: string[]; decoys: string[] }> = {};
let n = 0;
for (const Q of QUERIES) {
  const e = raw.find((x: any) => x.slug === Q.slug && x.q === Q.q);
  const best: Record<string, number> = {};
  for (const r of e.reps) for (const [id, p] of Object.entries(r.scores) as [string, number][]) best[id] = Math.max(best[id] ?? 0, p);
  const fires = e.reps.some((r: any) => Math.max(...(Object.values(r.scores) as number[])) < 0.7);
  const cands = Object.entries(best).filter(([, p]) => p >= 0.45).map(([id]) => id);
  if (!fires || cands.length === 0) continue;
  const low = Object.entries(best).filter(([, p]) => p <= 0.3).map(([id]) => id).sort();
  for (let i = low.length - 1; i > 0; i--) { const j = Math.floor(rnd() * (i + 1)); [low[i], low[j]] = [low[j]!, low[i]!]; }
  const decoys = low.slice(0, Math.max(3, cands.length));
  key[`${Q.slug}|${Q.q}`] = { candidates: cands, decoys };
  const show = new Set([...cands, ...decoys]);
  const asked = quickBlocks(load(Q.slug));
  md += `\n\n## ${Q.slug} | ${Q.q}\n\nThe reader typed: **${Q.q}**\n\n`;
  for (const b of asked) if (show.has(b.id)) { n++; md += `- \`${b.id}\` ${b.text.slice(0, 1500).replace(/\s+/g, " ")}${b.text.length > 1500 ? " […]" : ""}\n`; }
}
writeFileSync(`${DIR}/pool-abstract.md`, md);
writeFileSync(`${DIR}/pool-abstract-key.json`, JSON.stringify(key, null, 1));
console.log(`${Object.keys(key).length} queries, ${n} blocks to judge`);
for (const [k, v] of Object.entries(key)) console.log(`  ${k}: ${v.candidates.length} candidates + ${v.decoys.length} decoys`);
process.exit(0);
