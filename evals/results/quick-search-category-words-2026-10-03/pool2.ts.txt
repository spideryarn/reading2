/* Second blind pool (Sol F2): an independent judge, with each paragraph's
   section heading shown. Per intent (a base query and its bare-label variants):
   every paragraph scoring >= 0.45 on any run where nothing cleared 0.7, plus a
   third as many decoys (never above 0.3 on the base query). Also the two
   absent-topic queries whose fallback shows anything. Article order, no scores.
   Prints article text: scratch only. */
import { readFileSync, writeFileSync } from "node:fs";
import { DIR, quickBlocks } from "./lib.ts";
import { QUERIES, load } from "./abstract-run.ts";
import { VARIANTS } from "./round2-run.ts";

const raw = JSON.parse(readFileSync(`${DIR}/abstract-raw.json`, "utf8"));
const r2 = JSON.parse(readFileSync(`${DIR}/round2-raw.json`, "utf8"));
let seed = 20261003;
const rnd = () => ((seed = (seed * 1103515245 + 12345) & 0x7fffffff) / 0x7fffffff);
const top = (r: any) => Math.max(...(Object.values(r.scores) as number[]));

type Intent = { slug: string; base: string; entries: any[] };
const intents: Intent[] = [];
for (const Q of QUERIES) {
  const e = raw.find((x: any) => x.slug === Q.slug && x.q === Q.q);
  const vs = VARIANTS.filter((v) => v.slug === Q.slug && v.base === Q.q && v.how !== "request").map((v) => r2.find((x: any) => x.slug === v.slug && x.q === v.q));
  intents.push({ slug: Q.slug, base: Q.q, entries: [e, ...vs] });
}
for (const h of r2.filter((x: any) => x.set === "hard")) intents.push({ slug: h.slug, base: h.q, entries: [h] });

let md = "";
const key: Record<string, { candidates: string[]; decoys: string[] }> = {};
let n = 0;
for (const I of intents) {
  const cands = new Set<string>();
  for (const e of I.entries) for (const r of e.reps) if (top(r) < 0.7) for (const [id, p] of Object.entries(r.scores) as [string, number][]) if (p >= 0.45) cands.add(id);
  if (!cands.size) continue;
  const best: Record<string, number> = {};
  for (const e of I.entries) for (const r of e.reps) for (const [id, p] of Object.entries(r.scores) as [string, number][]) best[id] = Math.max(best[id] ?? 0, p);
  const low = Object.entries(best).filter(([, p]) => p <= 0.3).map(([id]) => id).sort();
  for (let i = low.length - 1; i > 0; i--) { const j = Math.floor(rnd() * (i + 1)); [low[i], low[j]] = [low[j]!, low[i]!]; }
  const decoys = low.slice(0, Math.max(2, Math.ceil(cands.size / 3)));
  key[`${I.slug}|${I.base}`] = { candidates: [...cands], decoys };
  const show = new Set([...cands, ...decoys]);
  const blocks = load(I.slug);
  const askable = new Set(quickBlocks(blocks).map((b: any) => b.id));
  md += `\n\n## ${I.slug} | ${I.base}\n\nThe reader typed: **${I.base}**\n\n`;
  let heading = "(start of the article)";
  for (const b of blocks) {
    if (b.kind === "heading") { heading = b.text.replace(/\s+/g, " ").slice(0, 120); continue; }
    if (!askable.has(b.id) || !show.has(b.id)) continue;
    n++;
    md += `- \`${b.id}\` [under the heading: “${heading}”] ${b.text.slice(0, 1500).replace(/\s+/g, " ")}${b.text.length > 1500 ? " […]" : ""}\n`;
  }
}
writeFileSync(`${DIR}/pool2.md`, md);
writeFileSync(`${DIR}/pool2-key.json`, JSON.stringify(key, null, 1));
console.log(`${Object.keys(key).length} intents, ${n} blocks to judge`);
for (const [k, v] of Object.entries(key)) console.log(`  ${k}: ${v.candidates.length} candidates + ${v.decoys.length} decoys`);
process.exit(0);
