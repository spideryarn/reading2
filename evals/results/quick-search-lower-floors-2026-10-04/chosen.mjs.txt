/* 261004: the chosen rule (0.65, fallback 0.4, best 8) against today's (0.7,
   0.5, 8): the specifics the write-up quotes. No model calls. */
import { writeFileSync } from "node:fs";
import { entries, FIXTURES, rule, sorted, verdict } from "./replay-floors.mjs";
const OUT = process.env.QOUT ?? "data/qeval";
const out = {};
for (const [name, M, F] of [["today", 0.7, 0.5], ["chosen", 0.65, 0.4]]) {
  const o = (out[name] = { main: M, fallback: F });
  /* Absent and near-miss topics that show anything: list sizes per run. */
  o.absentLists = {};
  for (const e of entries.filter((x) => x.group === "absent")) { const sizes = e.reps.map((r) => rule(M, F, 8)(sorted(r.scores, e.slug)).ids.length); if (sizes.some(Boolean)) o.absentLists[`${e.slug.slice(0, 12)}|${e.q}`] = sizes; }
  /* Fallback lists on the bare set, fixtures (all judged). */
  const t = { lists: 0, shown: 0, right: 0, topRight: 0, nothingRight: 0 };
  for (const e of entries.filter((x) => x.group === "bare" && FIXTURES.includes(x.slug))) for (const r of e.reps) {
    const { ids, fb } = rule(M, F, 8)(sorted(r.scores, e.slug)); if (!fb || !ids.length) continue;
    const v = ids.map(([id]) => verdict(e, id));
    t.lists++; t.shown += ids.length; t.right += v.filter((x) => x === "y").length; if (v[0] === "y") t.topRight++; if (!v.includes("y")) t.nothingRight++;
  }
  o.bareFixtureFallbackLists = t;
  /* How many runs differ from today's list, per group, all six articles. */
  o.runsChanged = {};
  for (const g of ["working", "bare", "request", "absent"]) { let ch = 0, n = 0; for (const e of entries.filter((x) => x.group === g)) for (const r of e.reps) { n++; const s = sorted(r.scores, e.slug); if (rule(0.7, 0.5, 8)(s).ids.map(([id]) => id).join() !== rule(M, F, 8)(s).ids.map(([id]) => id).join()) ch++; } o.runsChanged[g] = `${ch} of ${n}`; }
  /* Queries crossing the ordinary floor between runs, including empty runs.
     This measures the boundary, not how severe the list-size cliff is. */
  let flip = 0, q = 0; for (const e of entries.filter((x) => x.group !== "absent")) { q++; if (new Set(e.reps.map((r) => Object.values(r.scores).some((p) => p >= M))).size > 1) flip++; } o.queriesFlippingMode = `${flip} of ${q}`;
  const spreads = entries.filter((e) => e.group !== "absent").map((e) => {
    const sizes = e.reps.map((r) => rule(M, F, 8)(sorted(r.scores, e.slug)).ids.length);
    return Math.max(...sizes) - Math.min(...sizes);
  });
  o.listSizeSpread = { queries: spreads.length, mean: spreads.reduce((a, b) => a + b, 0) / spreads.length, max: Math.max(...spreads) };
}
/* Per query, sizes today -> chosen, for the bare set and Greg's article. */
out.perQuery = entries.filter((x) => x.group === "bare" || x.q === "human memory").map((e) => ({ article: e.slug.slice(0, 12), typed: e.q, sizes: e.reps.map((r) => { const s = sorted(r.scores, e.slug); return `${rule(0.7, 0.5, 8)(s).ids.length}->${rule(0.65, 0.4, 8)(s).ids.length}`; }).join("  "), best: e.reps.map((r) => sorted(r.scores, e.slug)[0][1].toFixed(2)).join(" ") }));
let mx = 0; for (const e of entries.filter((x) => x.src === "near-miss")) for (const r of e.reps) mx = Math.max(mx, ...Object.values(r.scores));
out.nearMissBestScore = mx;
console.log(JSON.stringify({ today: out.today, chosen: out.chosen, nearMissBestScore: mx }, null, 1));
for (const p of out.perQuery) console.log(`${p.article} | ${p.typed.padEnd(16)} ${p.sizes}   best ${p.best}`);
writeFileSync(`${OUT}/chosen.json`, JSON.stringify(out, null, 1));
